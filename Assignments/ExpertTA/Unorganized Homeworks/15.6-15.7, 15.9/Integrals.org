from sympy import *

def inline(s):
    return "\[" + s + "\]\n"

x, h, R_1, R_2 = symbols('x h R_1 R_2', real=True, positive=True)

integrand1 = x**2 / (x**2 + h**2)**Rational(3,2)
result1 = simplify(integrate(integrand1, (x, R_1, R_2)))


inline(
    fr"\int_{{{latex(R_1)}}}^{{{latex(R_2)}}} "
    fr"{latex(integrand1)} \, dx = {latex(result1)}"
)

#+END_SRC
